LineTerminator = \R
InputCharacter = .

DecIntegerLiteral = \d+

WhiteSpace = {LineTerminator} | \s+

Identificatore = [:letter:]([:letter:] | [:letterdigit:])*

"true" | "false" { return sf.newSymbol("BOOL", ParserSym.BOOL, yytext()); } //bool?
"="         { return sf.newSymbol("=", ParserSym.EQUAL); }
